显存需求估算


def estimate_gpu_memory(model_params, batch_size, seq_len):
    """
    估算 GPU 显存需求

    参数:
        model_params: 模型参数量 (B)
        batch_size: 批次大小
        seq_len: 序列长度
    """
    # 模型权重 (FP16)
    model_memory = model_params * 2  # 2 bytes per param

    # 优化器状态 (Adam: 8 bytes per param)
    optimizer_memory = model_params * 8

    # 梯度 (FP16)
    gradient_memory = model_params * 2

    # 激活值 (近似)
    activation_memory = batch_size * seq_len * 1024 * 2

    total = model_memory + optimizer_memory + gradient_memory + activation_memory
    return total / (1024**3)  # GB

# 示例：7B 模型训练
mem = estimate_gpu_memory(7e9, 32, 512)
print(f"预计显存需求：{mem:.1f} GB")

云 vs 自建

云服务

  • ✅ 按需付费
  • ✅ 弹性扩展
  • ❌ 长期成本高

自建

  • ✅ 长期成本低
  • ✅ 数据可控
  • ❌ 前期投入大

# 成本对比示例
cloud_monthly = 20 * 24 * 30 * 0.5  # $0.5/小时，20 卡
self_build = (50000 * 20 + 5000) / 36  # 硬件/3 年折旧

print(f"云：${cloud_monthly:.0f}/月")
print(f"自建：${self_build:.0f}/月")
