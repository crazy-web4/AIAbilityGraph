量化技术详解

量化类型

  • 训练后量化 (PTQ): 无需重新训练
  • 量化感知训练 (QAT): 训练时模拟量化

精度等级

  • FP32: 100% 精度，32 bit
  • FP16: ~100%, 16 bit
  • INT8: 99%+, 8 bit
  • INT4: 95%+, 4 bit

# PyTorch 动态量化
from torch.quantization import quantize_dynamic
import torch.nn as nn

# 只量化 Linear 层
quantized_model = quantize_dynamic(
    model,
    {nn.Linear},
    dtype=torch.qint8
)

# 保存量化模型
torch.save(quantized_model.state_dict(), "model_int8.pt")

INT8 量化实战


from torch.ao.quantization import (
    QuantStub, DeQuantStub,
    prepare_qat, convert,
    QConfig, default_qconfig
)

class QuantizedModel(nn.Module):
    def __init__(self):
        super().__init__()
        self.quant = QuantStub()
        self.fc1 = nn.Linear(768, 3072)
        self.fc2 = nn.Linear(3072, 768)
        self.dequant = DeQuantStub()

    def forward(self, x):
        x = self.quant(x)
        x = self.fc1(x)
        x = self.fc2(x)
        return self.dequant(x)

# 量化感知训练
model = QuantizedModel()
model.qconfig = QConfig(
    activation=default_qconfig.activation,
    weight=default_qconfig.weight
)
prepared_model = prepare_qat(model)

# 训练...
# 转换为量化模型
quantized_model = convert(prepared_model)

模型剪枝


import torch.nn.utils.prune as prune
import torch.nn as nn

model = nn.Linear(10, 10)

# L1 非结构化剪枝
prune.l1_unstructured(model, name='weight', amount=0.3)
# 剪掉 30% 权重

# 结构化剪枝（整行/整列）
prune.ln_structured(
    model,
    name='weight',
    amount=0.5,
    n=2,  # L2 范数
    dim=0  # 按行剪枝
)

# 永久移除剪枝
prune.remove(model, 'weight')

结构化剪枝：可以真正减少推理时间，非结构化剪枝只减少存储

知识蒸馏

核心思想：让小模型（学生）学习大模型（教师）的输出分布

$$\mathcal{L} = \alpha \cdot \text{CE}(y, \hat{y}) + (1-\alpha) \cdot T^2 \cdot \text{KL}(p_{teacher} || p_{student})$$ 

def distillation_loss(student_logits, teacher_logits,
                      true_labels, temperature=4, alpha=0.7):
    """知识蒸馏损失"""
    ce_loss = nn.CrossEntropyLoss()(student_logits, true_labels)

    soft_target = nn.Softmax(dim=1)(teacher_logits / temperature)
    student_log_prob = nn.LogSoftmax(dim=1)(student_logits / temperature)

    kl_loss = nn.KLDivLoss()(student_log_prob, soft_target)

    # 组合损失
    loss = alpha * ce_loss + (1 - alpha) * (temperature ** 2) * kl_loss
    return loss

GPTQ - 4-bit 量化


from transformers import AutoModelForCausalLM
from auto_gptq import AutoGPTQForCausalLM, BaseQuantizeConfig

# GPTQ 4-bit 量化配置
quantize_config = BaseQuantizeConfig(
    bits=4,
    group_size=128,
    desc_act=False  # 关闭 desc_act 加速推理
)

# 加载并量化
model = AutoGPTQForCausalLM.from_pretrained(
    "meta-llama/Llama-2-7b-hf",
    quantize_config
)

# 用校准数据量化
model.quantize(calibration_data, batch_size=1)

# 保存量化模型
model.save_quantized("llama-2-7b-gptq-4bit")

效果：7B 模型从 14GB 压缩到 4GB，速度提升 2-3 倍

AWQ - 激活感知量化

AWQ 原理：保护重要权重（高激活值）不被量化误差影响


from awq import AutoAWQForCausalLM

# AWQ 量化
quant_config = { "zero_point": True, "q_group_size": 128, "w_bit": 4, "version": "GEMM" }

model = AutoAWQForCausalLM.from_pretrained(
    "meta-llama/Llama-2-7b-hf"
)

# 量化
model.quantize(tokenizer, quant_config=quant_config)

# 保存
model.save_quantized("llama-2-7b-awq-4bit")
