装饰器与上下文管理器


# 装饰器示例：计时
import time
from functools import wraps

def timer(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} 耗时：{end-start:.3f}s")
        return result
    return wrapper

@timer
def train_model():
    # 训练代码
    pass

# 上下文管理器
from contextlib import contextmanager

@contextmanager
def timer_context(name):
    start = time.time()
    try:
        yield
    finally:
        print(f"{name} 耗时：{time.time()-start:.3f}s")

异步编程


import asyncio
import aiohttp

async def fetch(session, url):
    async with session.get(url) as response:
        return await response.text()

async def fetch_all(urls):
    async with aiohttp.ClientSession() as session:
        tasks = [fetch(session, url) for url in urls]
        results = await asyncio.gather(*tasks)
    return results

# 运行
urls = ["url1", "url2", "url3"]
results = asyncio.run(fetch_all(urls))

# async/await 在 AI API 调用中的应用

缓存

from functools import lru_cache

@lru_cache(maxsize=1000)
def expensive_compute(x, y):
    time.sleep(1)  # 模拟计算
    return x + y

类型提示


from typing import List, Dict, Optional, Callable, TypeVar

T = TypeVar('T')

def process_data(
    items: List[str],
    config: Optional[Dict[str, float]] = None,
    callback: Optional[Callable[[str], T]] = None
) -> List[T]:
    """带类型提示的函数"""
    results = []
    for item in items:
        if callback:
            results.append(callback(item))
    return results
