Docker 容器化


# Dockerfile
FROM python:3.10-slim

WORKDIR /app

# 安装依赖
RUN pip install torch torchvision \
    transformers fastapi uvicorn

# 复制代码
COPY . .

# 暴露端口
EXPOSE 8000

# 启动命令
CMD ["uvicorn", "main:app", "--host", "0.0.0.0"]
                

# 构建镜像
docker build -t ai-service .

# 运行容器
docker run -p 8000:8000 ai-services

# Docker Compose
docker-compose up -d

Kubernetes 部署


# deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: ai-service
spec:
  replicas: 3
  selector:
    matchLabels:
      app: ai-service
  template:
    metadata:
      labels:
        app: ai-service
    spec:
      containers:
      - name: ai-service
        image: ai-service:latest
        ports:
        - containerPort: 8000
        resources:
          requests:
            nvidia.com/gpu: 1
          limits:
            memory: 8Gi

性能监控


# Prometheus 指标
from prometheus_client import Counter, Histogram, start_http_server

REQUEST_COUNT = Counter('ai_requests', 'Total requests')
REQUEST_LATENCY = Histogram('ai_latency', 'Request latency')
ERROR_COUNT = Counter('ai_errors', 'Total errors')

def predict(input_data):
    REQUEST_COUNT.inc()

    start = time.time()
    try:
        result = model.predict(input_data)
        REQUEST_LATENCY.observe(time.time() - start)
        return result
    except Exception as e:
        ERROR_COUNT.inc()
        raise

# 启动指标服务器
