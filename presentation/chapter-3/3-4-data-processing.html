数据清洗技术


# 示例：文本清洗 pipeline
import re
from langdetect import detect

def clean_text(text):
    # 1. 移除 HTML 标签
    text = re.sub(r'<[^>]+>', '', text)

    # 2. 标准化空白字符
    text = re.sub(r'\s+', ' ', text).strip()

    # 3. 移除过短行
    lines = text.split('\n')
    lines = [l for l in lines if len(l) > 10]
    text = '\n'.join(lines)

    # 4. 语言过滤
    try:
        if detect(text) != 'en':
            return None
    except:
        return None

    return text

# 批量处理
from datasets import load_dataset

数据质量评估

指标 描述 阈值
困惑度 (Perplexity) 语言模型分数 < 1000
重复率 n-gram 重复比例 < 10%
特殊字符比 非字母数字占比 < 30%
平均句长 句子平均长度 10-100 词 

# 质量评分示例
def quality_score(text, ref_model):
    """用语言模型计算质量分数"""
    perplexity = ref_model.perplexity(text)

    # 特殊字符比例
    special_ratio = len(re.findall(r'[^a-zA-Z0-9\s]', text)) / len(text)

    # n-gram 重复率
    ngrams = [text[i:i+5] for i in range(len(text)-4)]
    dup_ratio = len(ngrams) - len(set(ngrams)) / len(ngrams)

    # 综合评分
    score = (perplexity < 1000) and (special_ratio < 0.3) and (dup_ratio < 0.1)
    return score
