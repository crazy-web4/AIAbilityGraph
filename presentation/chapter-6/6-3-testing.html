单元测试示例


# test_model.py
import pytest
import torch
from model import MyModel

def test_model_output_shape():
    model = MyModel()
    x = torch.randn(4, 10, 512)  # batch=4, seq=10, dim=512
    output = model(x)

    assert output.shape == (4, 10, 512)
    assert not torch.isnan(output).any()

def test_model_deterministic():
    model = MyModel()
    model.eval()

    x = torch.randn(2, 5, 512)

    with torch.no_grad():
        out1 = model(x)
        out2 = model(x)

    assert torch.allclose(out1, out2)

# 运行：pytest test_model.py -v

# test_data.py
import pandas as pd

def test_data_quality(data_df):
    """数据质量测试"""
    # 无空值
    assert data_df.isnull().sum().sum() == 0

    # 数值范围合理
    assert (data_df['age'] >= 0).all()
    assert (data_df['age'] <= 120).all()

    # 类别值在预期内
    expected_categories = ['A', 'B', 'C']
    assert set(data_df['category'].unique()).issubset(expected_categories)

    )

def test_latency_regression(model, input_data):
    """延迟回归测试"""
    import time

    start = time.time()
    _ = model(input_data)
    latency = (time.time() - start) * 1000

    # 允许 20% 波动
    assert latency <= BASELINE_LATENCY * 1.2, (
        f"延迟增加过多：{latency}ms"
    )
